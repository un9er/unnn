from flask import Flask, request, jsonify, render_template import requests import logging import os app = Flask(__name__) logging.basicConfig(level=logging.INFO) # Получаем Telegram Bot Token из переменных окружения TELEGRAM_BOT_TOKEN = os.environ.get("TELEGRAM_BOT_TOKEN") if not TELEGRAM_BOT_TOKEN: logging.error("TELEGRAM_BOT_TOKEN не задан в переменных окружения") raise Exception("TELEGRAM_BOT_TOKEN обязателен") TELEGRAM_API_URL = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}" @app.route("/") def index(): """ Главная страница сканера. Требуется передать параметр chat_id в query string. Пример: https://yourdomain.com/?chat_id=123456789 """ chat_id = request.args.get("chat_id") if not chat_id: return "Не указан параметр chat_id", 400 return render_template("scanner.html", chat_id=chat_id) @app.route("/result", methods=["POST"]) def result(): """ Получает результат сканирования (JSON с chat_id и result), отправляет сообщение в Telegram через Bot API. """ data = request.get_json() if not data: return jsonify({"error": "Нет переданных данных"}), 400 chat_id = data.get("chat_id") qr_result = data.get("result") if not chat_id or not qr_result: return jsonify({"error": "Параметры chat_id и result обязательны"}), 400 send_url = f"{TELEGRAM_API_URL}/sendMessage" payload = { "chat_id": chat_id, "text": f"Результат сканирования QR-кода:\n{qr_result}" } response = requests.post(send_url, json=payload) if response.status_code != 200: logging.error(f"Ошибка отправки сообщения: {response.text}") return jsonify({"error": "Ошибка отправки сообщения в Telegram"}), 500 return jsonify({"status": "success"}), 200 if __name__ == "__main__": # Для разработки – запуск на 0.0.0.0:5000 с отладкой app.run(host="0.0.0.0", port=5000, debug=True)
